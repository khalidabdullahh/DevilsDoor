using System.Collections;
using UnityEngine;

// ============================================================================
// DEVIL'S DOOR — Heavy Tank Enemy AI
// Character: #03 Oni Guard
// Role: Massive Area Denial, Uninterruptible Hyper-Armor Brute
// ============================================================================
public class OniGuardAI : MonoBehaviour, IDamageable
{
    [Header("Health & Hyper-Armor (Paharer moto bishal shokti)")]
    public int maxHealth = 1600;
    private int currentHealth;
    public bool isHyperArmorActive = false; // Windup e kokhono attack interrupt hobe na

    [Header("Kanabo Melee Slam (Bishal Lohar Gada)")]
    public float slamRadius = 3.5f;
    public int slamDamage = 140;
    public LayerMask playerLayer;
    public Transform slamImpactPoint;

    [Header("Camera & Hit-Stop Feedback")]
    public float hitStopTimeDilation = 0.05f;
    public float hitStopDuration = 0.08f;

    private Transform targetPlayer;
    private bool isAttacking = false;

    void Start()
    {
        currentHealth = maxHealth;
        GameObject player = GameObject.FindGameObjectWithTag("Player");
        if (player != null) targetPlayer = player.transform;

        StartCoroutine(BruteBehaviorLoop());
    }

    void Update()
    {
        // 2.5D position lock
        Vector3 p = transform.position;
        p.z = 0f;
        transform.position = p;
    }

    private IEnumerator BruteBehaviorLoop()
    {
        while (currentHealth > 0)
        {
            if (targetPlayer == null) yield return new WaitForSeconds(0.5f);

            float dist = Vector3.Distance(transform.position, targetPlayer.position);

            // Player kache thakle heavy kanabo attack shuru hobe
            if (dist < 4.0f && !isAttacking)
            {
                yield return StartCoroutine(KanaboSlamAttack());
            }
            else if (!isAttacking)
            {
                // Dhire dhire player er dike hete jaoa
                float step = 2.2f * Time.deltaTime;
                Vector3 target = new Vector3(targetPlayer.position.x, transform.position.y, 0);
                transform.position = Vector3.MoveTowards(transform.position, target, step);
            }

            yield return null;
        }
    }

    private IEnumerator KanaboSlamAttack()
    {
        isAttacking = true;
        isHyperArmorActive = true; // Windup shuru, no flinch

        // Anticipation / Windup (0.8s)
        yield return new WaitForSeconds(0.8f);

        // Active Impact Frame
        TriggerGroundQuakeImpact();

        // Hit-stop screen freeze logic
        yield return StartCoroutine(ApplyEngineHitStop());

        // Recovery window (0.7s)
        isHyperArmorActive = false;
        yield return new WaitForSeconds(0.7f);

        isAttacking = false;
    }

    private void TriggerGroundQuakeImpact()
    {
        Vector3 impactPos = slamImpactPoint != null ? slamImpactPoint.position : transform.position;
        Collider[] victims = Physics.OverlapSphere(impactPos, slamRadius, playerLayer);

        foreach (var col in victims)
        {
            var target = col.GetComponent<IDamageable>();
            if (target != null)
            {
                target.TakeDamage(slamDamage, transform.position);
                Debug.Log("Oni Guard er Kanabo slam e player chhitke geche!");
            }
        }
    }

    private IEnumerator ApplyEngineHitStop()
    {
        // Heavy impact feel anar jonno time dilation
        Time.timeScale = hitStopTimeDilation;
        yield return new WaitForSecondsRealtime(hitStopDuration);
        Time.timeScale = 1.0f;
    }

    public void TakeDamage(int damage, Vector3 attackerPosition)
    {
        if (isHyperArmorActive)
        {
            // Damage nibe kintu attack bondho hobe na
            currentHealth -= (int)(damage * 0.85f);
            Debug.Log("Oni Guard Hyper-Armor diye attack push korche! Flinch hoy ni.");
        }
        else
        {
            currentHealth -= damage;
        }

        if (currentHealth <= 0)
        {
            Debug.Log("Oni Guard Dhuloy pore gelo!");
            Destroy(gameObject, 0.5f);
        }
    }
}
