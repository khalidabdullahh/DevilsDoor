using System.Collections;
using UnityEngine;

// ============================================================================
// DEVIL'S DOOR — Protagonist Controller
// Character: #01 Shadow Ninja
// Role: 2.5D Agile Playable Character
// ============================================================================
[RequireComponent(typeof(CharacterController))]
public class ShadowNinjaController : MonoBehaviour
{
    [Header("Movement Settings (Chonchol Movement)")]
    // Shadharon daurano ebong batashe control speed
    public float runSpeed = 7.5f;
    public float jumpForce = 12.5f;
    public float gravity = -28.0f;
    public float zAxisLockPosition = 0f; // 2.5D plane e Z-axis lock rakhar jonno

    [Header("Dash Settings (Chaya Dash)")]
    public float dashSpeed = 16.0f;
    public float dashDuration = 0.22f;
    public float dashCooldown = 0.8f;
    private float lastDashTime = -999f;
    private bool isDashing = false;

    [Header("Dynamic Scarf Settings (Rongin Scarf Physics)")]
    // Dynamic crimson scarf er bone chain target
    public Transform[] scarfBones;
    public float scarfDrag = 0.85f;
    public float scarfTension = 0.45f;
    private Vector3[] scarfVelocities;
    private Vector3[] scarfPositions;

    [Header("Combat & Frame Data (Katana Attack System)")]
    public Transform katanaHitboxCenter;
    public Vector3 hitboxHalfExtents = new Vector3(0.9f, 0.6f, 0.4f);
    public LayerMask enemyLayer;
    public int attackDamage = 65;
    
    // Attack combo state tracking
    private int comboStep = 0;
    private bool isAttacking = false;
    private float comboResetTimer = 0f;
    private const float COMBO_EXPIRATION = 0.9f;

    // Components & internal state
    private CharacterController controller;
    private Vector3 velocity;
    private bool isGrounded;
    private float horizontalInput;

    void Awake()
    {
        // Component initialize kora
        controller = GetComponent<CharacterController>();
        InitializeScarfChain();
    }

    void Update()
    {
        // 2.5D plane e Z-axis clamp kora jate character line theke ber na hoy
        ClampToDepthPlane();

        // Dash cholakalin shadharon input bondho thakbe
        if (isDashing) return;

        HandleGroundCheck();
        HandleInput();
        HandleMovement();
        HandleCombatInput();
        UpdateScarfPhysics();
    }

    // 2.5D Depth enforcement logic
    private void ClampToDepthPlane()
    {
        Vector3 pos = transform.position;
        if (Mathf.Abs(pos.z - zAxisLockPosition) > 0.001f)
        {
            pos.z = zAxisLockPosition;
            transform.position = pos;
        }
    }

    private void HandleGroundCheck()
    {
        isGrounded = controller.isGrounded;
        if (isGrounded && velocity.y < 0)
        {
            // Floor e smooth thakar jonno choto downward force
            velocity.y = -2f;
        }
    }

    private void HandleInput()
    {
        // 2.5D horizontal input (X-axis)
        horizontalInput = Input.GetAxisRaw("Horizontal");

        // Character er face ghuranor logic
        if (horizontalInput > 0.05f)
        {
            transform.rotation = Quaternion.Euler(0, 90, 0);
        }
        else if (horizontalInput < -0.05f)
        {
            transform.rotation = Quaternion.Euler(0, -90, 0);
        }

        // Jump trigger
        if (Input.GetButtonDown("Jump") && isGrounded && !isAttacking)
        {
            velocity.y = jumpForce;
        }

        // Dash trigger
        if (Input.GetKeyDown(KeyCode.LeftShift) && Time.time >= lastDashTime + dashCooldown)
        {
            StartCoroutine(PerformShadowDash());
        }
    }

    private void HandleMovement()
    {
        // Shadharon horizontal daurano
        Vector3 move = new Vector3(horizontalInput, 0, 0) * runSpeed;
        controller.Move(move * Time.deltaTime);

        // Gravity apply kora
        velocity.y += gravity * Time.deltaTime;
        controller.Move(velocity * Time.deltaTime);
    }

    // Shadow Air Dash: invincible frames ebong speed boost
    private IEnumerator PerformShadowDash()
    {
        isDashing = true;
        lastDashTime = Time.time;
        float direction = transform.forward.z > 0 ? 1f : -1f; // Facing onushare dash
        float elapsed = 0f;

        // Gravity temporary off kora dash er somoy
        velocity.y = 0f;

        while (elapsed < dashDuration)
        {
            Vector3 dashMove = new Vector3(direction * dashSpeed, 0, 0);
            controller.Move(dashMove * Time.deltaTime);
            elapsed += Time.deltaTime;
            yield return null;
        }

        isDashing = false;
    }

    private void HandleCombatInput()
    {
        // Combo expiration timer
        if (comboStep > 0)
        {
            comboResetTimer += Time.deltaTime;
            if (comboResetTimer > COMBO_EXPIRATION)
            {
                comboStep = 0;
            }
        }

        // Primary Attack Button
        if (Input.GetKeyDown(KeyCode.J) && !isAttacking)
        {
            StartCoroutine(ExecuteAttackFrameData());
        }
    }

    // Precise frame-data based katana combo
    private IEnumerator ExecuteAttackFrameData()
    {
        isAttacking = true;
        comboStep = (comboStep % 3) + 1;
        comboResetTimer = 0f;

        // Phase 1: Startup (Frames 1-3 ~ 0.05s) - Input buffer locked
        yield return new WaitForSeconds(0.05f);

        // Phase 2: Active Hitbox Window (Frames 4-6 ~ 0.05s)
        DetectMeleeCollisions();
        yield return new WaitForSeconds(0.05f);

        // Phase 3: Recovery Window (Frames 7-10 ~ 0.08s)
        yield return new WaitForSeconds(0.08f);

        isAttacking = false;
    }

    private void DetectMeleeCollisions()
    {
        if (katanaHitboxCenter == null) return;

        // 2.5D box sweep diye enemy hit detect kora
        Collider[] hits = Physics.OverlapBox(
            katanaHitboxCenter.position,
            hitboxHalfExtents,
            katanaHitboxCenter.rotation,
            enemyLayer
        );

        foreach (Collider hit in hits)
        {
            Debug.Log("Enemy hit hoise: " + hit.name);
            // Hit target er component call kora
            var damageable = hit.GetComponent<IDamageable>();
            if (damageable != null)
            {
                damageable.TakeDamage(attackDamage, transform.position);
            }
        }
    }

    // Dynamic Crimson Scarf er Verlet Physics setup
    private void InitializeScarfChain()
    {
        if (scarfBones == null || scarfBones.Length == 0) return;

        scarfPositions = new Vector3[scarfBones.Length];
        scarfVelocities = new Vector3[scarfBones.Length];

        for (int i = 0; i < scarfBones.Length; i++)
        {
            scarfPositions[i] = scarfBones[i].position;
            scarfVelocities[i] = Vector3.zero;
        }
    }

    private void UpdateScarfPhysics()
    {
        if (scarfBones == null || scarfBones.Length == 0) return;

        // Root bone player er ghare fixed thakbe
        scarfPositions[0] = scarfBones[0].position;

        // Dynamic tail solver
        for (int i = 1; i < scarfBones.Length; i++)
        {
            Vector3 target = scarfPositions[i - 1] - (transform.forward * 0.15f) + (Vector3.down * 0.08f);
            scarfPositions[i] = Vector3.Lerp(scarfPositions[i], target, scarfTension * Time.deltaTime * 60f);
            scarfBones[i].position = scarfPositions[i];
        }
    }
}

// Global damage interface
public interface IDamageable
{
    void TakeDamage(int damage, Vector3 attackerPosition);
}
