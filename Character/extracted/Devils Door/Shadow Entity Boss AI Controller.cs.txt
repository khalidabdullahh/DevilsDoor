using System.Collections;
using UnityEngine;

// ============================================================================
// DEVIL'S DOOR — Late-Game Dimensional Threat AI
// Character: #06 Shadow Entity / Devil's Servant
// Role: Living Negative Matter Void, Orbiting Obsidian Armor Shards
// ============================================================================
public class ShadowEntityAI : MonoBehaviour, IDamageable
{
    [Header("Void Core & Obsidian Shards (Obosthob Shorkar)")]
    public int maxHealth = 2400;
    private int currentHealth;
    public Transform[] obsidianShards;
    public float shardOrbitSpeed = 90f;
    public float shardRadius = 2.4f;

    [Header("Core Vulnerability State")]
    public bool isCoreExposed = false; // Carapace venge gele core vulnerable hobe
    public GameObject crimsonSingularityVFX;

    [Header("Dimensional Phase Attack")]
    public float phaseCooldown = 7.0f;
    private float lastPhaseTime = 0f;

    void Start()
    {
        currentHealth = maxHealth;
        StartCoroutine(BossCombatRoutine());
    }

    void Update()
    {
        // 2.5D position lock
        Vector3 pos = transform.position;
        pos.z = 0f;
        transform.position = pos;

        // Floating disconnected obsidian plates orbit
        UpdateObsidianCarapace();
    }

    private void UpdateObsidianCarapace()
    {
        if (obsidianShards == null || obsidianShards.Length == 0) return;

        float stepAngle = 360f / obsidianShards.Length;
        for (int i = 0; i < obsidianShards.Length; i++)
        {
            if (obsidianShards[i] == null) continue;

            float ang = (Time.time * shardOrbitSpeed) + (i * stepAngle);
            float rad = ang * Mathf.Deg2Rad;

            // Non-uniform jagged orbit
            Vector3 offset = new Vector3(Mathf.Cos(rad) * shardRadius, Mathf.Sin(rad) * (shardRadius * 0.7f), 0);
            obsidianShards[i].position = transform.position + offset;
        }
    }

    private IEnumerator BossCombatRoutine()
    {
        while (currentHealth > 0)
        {
            // Singularity Pulse attack
            yield return new WaitForSeconds(phaseCooldown);
            yield return StartCoroutine(TriggerSingularityCollapse());
        }
    }

    private IEnumerator TriggerSingularityCollapse()
    {
        Debug.Log("Shadow Entity: Crimson Gravitational Void shuru holo!");
        isCoreExposed = true; // Ei 2 second e player core e hit korte parbe

        if (crimsonSingularityVFX != null)
            crimsonSingularityVFX.SetActive(true);

        yield return new WaitForSeconds(2.0f);

        if (crimsonSingularityVFX != null)
            crimsonSingularityVFX.SetActive(false);

        isCoreExposed = false;
        Debug.Log("Shadow Entity: Carapace abar bondho hoye gelo.");
    }

    public void TakeDamage(int damage, Vector3 attackerPosition)
    {
        if (!isCoreExposed)
        {
            // Shards thakakalin 90% damage resist hobe
            int reducedDamage = Mathf.Max(1, (int)(damage * 0.1f));
            currentHealth -= reducedDamage;
            Debug.Log($"Obsidian Carapace e hit laglo! Shudhu {reducedDamage} damage holo.");
        }
        else
        {
            // Exposed crimson core e full critical damage
            int critDamage = damage * 2;
            currentHealth -= critDamage;
            Debug.Log($"CRITICAL HIT! Exposed Core e {critDamage} damage laglo!");
        }

        if (currentHealth <= 0)
        {
            Debug.Log("DEVIL'S DOOR: Shadow Entity bibhrom hoye milaye gelo!");
            Destroy(gameObject, 0.5f);
        }
    }
}
