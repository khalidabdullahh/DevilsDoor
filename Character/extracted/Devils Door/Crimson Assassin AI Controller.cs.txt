using System.Collections;
using UnityEngine;

// ============================================================================
// DEVIL'S DOOR — Fast Elite Stalker AI
// Character: #05 Crimson Assassin
// Role: High-Speed Wall-Bouncing Flanker, Dual Trench Daggers
// ============================================================================
public class CrimsonAssassinAI : MonoBehaviour, IDamageable
{
    [Header("Stats & Agility (Churanto Choncholota)")]
    public int maxHealth = 550;
    private int currentHealth;
    public float sprintSpeed = 9.2f;
    public float wallBounceVelocity = 14f;

    [Header("Twin Dagger Flurry (Dui Chhuri Hamla)")]
    public int daggerDamage = 35; // Per strike
    public float strikeRange = 1.6f;
    public LayerMask targetLayer;

    private Transform playerTarget;
    private bool isExecutingFlank = false;

    void Start()
    {
        currentHealth = maxHealth;
        GameObject p = GameObject.FindGameObjectWithTag("Player");
        if (p != null) playerTarget = p.transform;

        StartCoroutine(AssassinStalkerCycle());
    }

    void Update()
    {
        // 2.5D lock
        Vector3 pos = transform.position;
        pos.z = 0f;
        transform.position = pos;
    }

    private IEnumerator AssassinStalkerCycle()
    {
        while (currentHealth > 0)
        {
            if (playerTarget == null) yield return new WaitForSeconds(0.4f);

            float dist = Mathf.Abs(transform.position.x - playerTarget.position.x);

            // Player kache thakle direct confrontation na kore pechone jabe (Flank)
            if (dist < 3.0f && !isExecutingFlank)
            {
                yield return StartCoroutine(ExecuteFlankAndStab());
            }
            else if (!isExecutingFlank)
            {
                // Agile sprint approach
                float step = sprintSpeed * Time.deltaTime;
                Vector3 target = new Vector3(playerTarget.position.x, transform.position.y, 0);
                transform.position = Vector3.MoveTowards(transform.position, target, step);
            }

            yield return null;
        }
    }

    // Player er mathar upor diye jhap diye pechone churi boshano
    private IEnumerator ExecuteFlankAndStab()
    {
        isExecutingFlank = true;

        // Leap behind player
        float behindX = playerTarget.position.x + (playerTarget.forward.z > 0 ? -2.2f : 2.2f);
        Vector3 flankTarget = new Vector3(behindX, transform.position.y + 2.5f, 0);

        float t = 0f;
        Vector3 start = transform.position;
        while (t < 0.25f)
        {
            transform.position = Vector3.Lerp(start, flankTarget, t / 0.25f);
            t += Time.deltaTime;
            yield return null;
        }

        // Rapid 4-hit dagger flurry
        for (int i = 0; i < 4; i++)
        {
            Collider[] hits = Physics.OverlapSphere(transform.position, strikeRange, targetLayer);
            foreach (var h in hits)
            {
                var dmg = h.GetComponent<IDamageable>();
                if (dmg != null) dmg.TakeDamage(daggerDamage, transform.position);
            }
            yield return new WaitForSeconds(0.06f);
        }

        isExecutingFlank = false;
        yield return new WaitForSeconds(1.0f); // Cooldown
    }

    public void TakeDamage(int damage, Vector3 attackerPosition)
    {
        currentHealth -= damage;
        if (currentHealth <= 0)
        {
            Debug.Log("Crimson Assassin chhayay dhole porlo!");
            Destroy(gameObject);
        }
    }
}
